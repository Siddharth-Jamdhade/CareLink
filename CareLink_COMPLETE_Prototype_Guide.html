How to use this guide: Each step has a grey command box (things to type in terminal) and purple code blocks (file contents to copy-paste). Create each file and paste the code exactly as shown.

    """
    Creates a database session for each API request.

    """
    db = SessionLocal()
    try:
        yield db
    finally:

    original_filename = Column(String)

    # Relationships
    patient = relationship("Patient", back_populates="records")
    facts = relationship("ExtractedFact", back_populates="record")

    __tablename__ = "extracted_facts"

def extract_text_from_image(image_path: str) -> str:
    """
    Takes an image file path.
    Returns all text found in the image as a single string.
    """
    result = ocr_engine.ocr(image_path, cls=True)

    lines = []
    if result and result[0]:
        for line in result[0]:
            text = line[1][0]       # the detected text
            confidence = line[1][1] # how confident the OCR is
            if confidence > 0.5:    # only keep reasonably confident text
                lines.append(text)

    return "\n".join(lines)

11 AI Extraction Service (The Core)

📄 backend/app/services/llm_extractor.py
import json
from google import genai
from google.genai import types
from ..config import GEMINI_API_KEY

# Initialize the Gemini client
client = genai.Client(api_key=GEMINI_API_KEY)

EXTRACTION_PROMPT = """You are a medical record data entry assistant. You extract ONLY administrative facts from maternal health records. You NEVER interpret clinical data, diagnose, score risk, or recommend treatment.

From this record image and its OCR text, extract the following fields into JSON:

{
  "visit_date": "YYYY-MM-DD or null if unclear",
  "facility_name": "name of hospital/clinic/centre or null",
  "document_type": "one of: ANC Card, Prescription, Lab Report, Ultrasound Report, Referral Note, Vaccination Record, Discharge Summary, Other",
  "gestational_week": "integer or null",
  "pending_items": ["list of pending follow-ups, tests, or referrals"],
  "provider_name": "doctor or nurse name if visible, or null",
  "next_appointment": "YYYY-MM-DD or null",
  "confidence_notes": "describe anything unclear or hard to read"
}

Rules:
- Only extract what is clearly written or printed. Do not guess.
- If a field is unreadable or ambiguous, set it to null and explain in confidence_notes.
- Do NOT extract blood pressure, weight, heart rate, lab values, or any clinical measurements.
- pending_items should only list administrative follow-ups (next visit, tests to get, referrals).
- Return ONLY valid JSON. No explanation text before or after.

OCR Text from this document:
"""


def extract_facts_from_image(image_path: str, ocr_text: str) -> dict:
    """
    Sends the image + OCR text to Gemini.
    Returns a dictionary of extracted administrative facts.
    """
    # Read the image file
    with open(image_path, "rb") as f:
        image_bytes = f.read()

    # Determine image type
    if image_path.lower().endswith(".png"):
        mime = "image/png"
    else:
        mime = "image/jpeg"

    # Build the prompt
    full_prompt = EXTRACTION_PROMPT + ocr_text

    try:
        # Call Gemini with both image and text
        response = client.models.generate_content(
            model="gemini-2.0-flash",
            contents=[
                types.Part.from_bytes(data=image_bytes, mime_type=mime),
                types.Part.from_text(text=full_prompt),
            ],
        )

        # Parse the JSON response
        response_text = response.text.strip()

        # Remove markdown code fences if present
        if response_text.startswith("```"):
            response_text = response_text.split("\n", 1)[1]
            response_text = response_text.rsplit("```", 1)[0]

        result = json.loads(response_text)

        # Calculate confidence score
        confidence = calculate_confidence(result, ocr_text)
        result["_confidence"] = confidence

        return result

    except json.JSONDecodeError:
        return {
            "visit_date": None,
            "facility_name": None,
            "document_type": "Other",
            "gestational_week": None,
            "pending_items": [],
            "provider_name": None,
            "next_appointment": None,
            "confidence_notes": "Failed to parse AI response",
            "_confidence": 0.3,
        }
    except Exception as e:
        return {
            "visit_date": None,
            "facility_name": None,
            "document_type": "Other",
            "gestational_week": None,
            "pending_items": [],
            "provider_name": None,
            "next_appointment": None,
            "confidence_notes": f"Error: {str(e)}",
            "_confidence": 0.1,
        }


def calculate_confidence(result: dict, ocr_text: str) -> float:
    """
    Calculates a confidence score (0.0 to 1.0) based on:
    - How many fields were extracted
    - Whether the confidence_notes mention issues
    """
    score = 0.5  # base score

    # Boost for each non-null field
    fields = ["visit_date", "facility_name", "document_type", "gestational_week"]
    for field in fields:
        if result.get(field) is not None:
            score += 0.1

    # Boost if extracted text matches OCR text
    if result.get("facility_name") and result["facility_name"].lower() in ocr_text.lower():
        score += 0.05

    # Penalty if confidence_notes has content (means something was unclear)
    notes = result.get("confidence_notes", "")
    if notes and notes != "null" and len(notes) > 5:
        score -= 0.15

    # Clamp between 0.1 and 0.98
    return max(0.1, min(0.98, round(score, 2)))

12 Conflict Detection Service

📄 backend/app/services/conflict_detector.py
from datetime import datetime, timedelta
from typing import List
from ..models import ExtractedFact


def detect_conflicts(new_fact_data: dict, existing_facts: List[ExtractedFact]) -> List[dict]:
    """
    Compares a newly extracted fact against all existing facts for the same patient.
    Returns a list of conflict descriptions.
    """
    conflicts = []

    new_date = parse_date(new_fact_data.get("visit_date"))
    new_facility = (new_fact_data.get("facility_name") or "").lower()
    new_doc_type = (new_fact_data.get("document_type") or "").lower()
    new_week = new_fact_data.get("gestational_week")

    for existing in existing_facts:
        ex_date = parse_date(existing.visit_date)
        ex_facility = (existing.facility_name or "").lower()
        ex_doc_type = (existing.document_type or "").lower()
        ex_week = existing.gestational_week

        # Rule 1: Date collision — same dates but different facilities
        if new_date and ex_date:
            day_diff = abs((new_date - ex_date).days)
            if day_diff <= 2 and new_facility != ex_facility and new_facility and ex_facility:
                conflicts.append({
                    "type": "date_collision",
                    "message": f"Visit on {new_fact_data['visit_date']} at '{new_fact_data['facility_name']}' "
                               f"is within 2 days of visit on {existing.visit_date} at '{existing.facility_name}'",
                    "existing_fact_id": existing.id,
                })

        # Rule 2: Gestational week goes backwards
        if new_week and ex_week and new_date and ex_date:
            if new_date > ex_date and new_week < ex_week:
                conflicts.append({
                    "type": "week_mismatch",
                    "message": f"Gestational week went from {ex_week} to {new_week} "
                               f"between {existing.visit_date} and {new_fact_data['visit_date']} — weeks should increase",
                    "existing_fact_id": existing.id,
                })

        # Rule 3: Duplicate document
        if (new_doc_type == ex_doc_type
                and new_facility == ex_facility
                and new_date and ex_date
                and abs((new_date - ex_date).days) <= 3):
            conflicts.append({
                "type": "possible_duplicate",
                "message": f"Possible duplicate: same document type '{new_fact_data['document_type']}' "
                           f"from same facility within 3 days",
                "existing_fact_id": existing.id,
            })

    return conflicts


def parse_date(date_str):
    """Safely parse a date string."""
    if not date_str:
        return None
    try:
        return datetime.strptime(date_str, "%Y-%m-%d")
    except (ValueError, TypeError):
        return None

import json
import os
from typing import List
from ..models import ExtractedFact

# Load the protocol file
PROTOCOL_PATH = os.path.join(os.path.dirname(__file__), "..", "data", "anc_protocol.json")

with open(PROTOCOL_PATH, "r") as f:
    ANC_PROTOCOL = json.load(f)


def check_protocol_gaps(gestational_week: int, existing_facts: List[ExtractedFact]) -> List[dict]:
    """
    Compares the patient's records against the ANC protocol.
    Returns a list of missing visits and missing items.
    """
    missing = []

    # Collect all text from existing facts for searching
    all_text = ""
    for fact in existing_facts:
        all_text += f" {fact.document_type or ''} {fact.pending_items or ''} {fact.facility_name or ''}"
    all_text = all_text.lower()

    visit_count = len([f for f in existing_facts if f.visit_date])

    for visit in ANC_PROTOCOL["visits"]:
        # Only check visits that should have happened by now
        if gestational_week >= visit["expected_by_week"]:

            # Check if this visit number exists
            if visit["visit_number"] > visit_count:
                missing.append({
                    "type": "missing_visit",
                    "label": visit["label"],
                    "expected_by_week": visit["expected_by_week"],
                    "message": f"{visit['label']} — expected by Week {visit['expected_by_week']}, no record found"
                })

            # Check each required item
            for item in visit["required_items"]:
                if item.lower() not in all_text:
                    missing.append({
                        "type": "missing_item",
                        "label": item,
                        "expected_by_week": visit["expected_by_week"],
                        "message": f"{item} — expected by Week {visit['expected_by_week']}, not found in records"
                    })

    return missing

router = APIRouter()


@router.post("/", response_model=PatientResponse)
def create_patient(patient: PatientCreate, db: Session = Depends(get_db)):

@router.get("/", response_model=list[PatientResponse])
def list_patients(db: Session = Depends(get_db)):
    """List all patients."""
    return db.query(Patient).all()


@router.get("/{patient_id}", response_model=PatientResponse)
def get_patient(patient_id: str, db: Session = Depends(get_db)):
    """Get a single patient."""
    patient = db.query(Patient).filter(Patient.id == patient_id).first()
    if not patient:
        raise HTTPException(status_code=404, detail="Patient not found")
    return patient


@router.get("/{patient_id}/journey")
def get_journey(patient_id: str, db: Session = Depends(get_db)):
    """
    Get the full consultation-ready journey for a patient.
    This is THE key endpoint — it powers the one-page consult view.
    """
    patient = db.query(Patient).filter(Patient.id == patient_id).first()
    if not patient:
        raise HTTPException(status_code=404, detail="Patient not found")

    # Get all facts, sorted by visit date
    facts = (
        db.query(ExtractedFact)
        .filter(ExtractedFact.patient_id == patient_id)
        .order_by(ExtractedFact.visit_date.asc())
        .all()
    )

    # Add source image path to each fact
    fact_responses = []
    for fact in facts:
        fr = FactResponse.model_validate(fact)
        fr.source_image_path = fact.record.source_image_path if fact.record else None
        fact_responses.append(fr)

    # Check protocol gaps
    missing = check_protocol_gaps(patient.gestational_week, facts)

    # Collect conflicts
    conflicts = [
        {"fact_id": f.id, "details": f.conflict_details}
        for f in facts
        if f.status == "conflict"
    ]

    return {
        "patient": PatientResponse.model_validate(patient),
        "facts": fact_responses,
        "missing_items": missing,
        "conflicts": conflicts,
    }


@router.get("/{patient_id}/qr")
def get_qr_code(patient_id: str, db: Session = Depends(get_db)):
    """Generate a QR code image for this patient."""

    if not patient:

16 API Routes — Record Upload & Extraction

📄 backend/app/routers/records.py
import os
import json
import uuid
from fastapi import APIRouter, Depends, UploadFile, File, HTTPException
from sqlalchemy.orm import Session
from ..database import get_db
from ..models import Record, ExtractedFact
from ..config import UPLOAD_DIR
from ..services.ocr_service import extract_text_from_image
from ..services.llm_extractor import extract_facts_from_image
from ..services.conflict_detector import detect_conflicts

router = APIRouter()


@router.post("/upload/{patient_id}")
async def upload_and_extract(
    patient_id: str,
    file: UploadFile = File(...),
    db: Session = Depends(get_db),
):
    """
    Upload a record image → run OCR → run AI extraction → save facts → detect conflicts.
    This is the main pipeline endpoint.
    """

    # Step 1: Save the uploaded image
    patient_dir = os.path.join(UPLOAD_DIR, patient_id)
    os.makedirs(patient_dir, exist_ok=True)

    filename = f"{uuid.uuid4()}_{file.filename}"
    file_path = os.path.join(patient_dir, filename)

    with open(file_path, "wb") as f:
        content = await file.read()
        f.write(content)

    # Step 2: Create a Record entry
    record = Record(
        patient_id=patient_id,
        source_image_path=file_path,
        original_filename=file.filename,
    )
    db.add(record)
    db.commit()
    db.refresh(record)

    # Step 3: Run OCR
    ocr_text = extract_text_from_image(file_path)

    # Step 4: Run AI extraction
    extracted = extract_facts_from_image(file_path, ocr_text)

    # Step 5: Save extracted fact to database
    pending_items = extracted.get("pending_items", [])
    if isinstance(pending_items, list):
        pending_items = json.dumps(pending_items)

    fact = ExtractedFact(
        record_id=record.id,
        patient_id=patient_id,
        visit_date=extracted.get("visit_date"),
        facility_name=extracted.get("facility_name"),
        document_type=extracted.get("document_type"),
        gestational_week=extracted.get("gestational_week"),
        pending_items=pending_items,
        provider_name=extracted.get("provider_name"),
        next_appointment=extracted.get("next_appointment"),
        confidence=extracted.get("_confidence", 0.5),
        status="unverified",
        verified_by='["document"]',
        confidence_notes=extracted.get("confidence_notes"),
    )
    db.add(fact)
    db.commit()
    db.refresh(fact)

    # Step 6: Check for conflicts with existing records
    existing_facts = (
        db.query(ExtractedFact)
        .filter(ExtractedFact.patient_id == patient_id)
        .filter(ExtractedFact.id != fact.id)
        .all()
    )

    conflicts = detect_conflicts(extracted, existing_facts)

    # If conflicts found, update status
    if conflicts:
        fact.status = "conflict"
        fact.conflict_details = json.dumps(conflicts)
        db.commit()

    return {
        "record_id": record.id,
        "fact_id": fact.id,
        "extracted_data": extracted,
        "confidence": fact.confidence,
        "status": fact.status,
        "conflicts": conflicts,
        "ocr_text": ocr_text,
        "source_image": file_path,
    }

def confirm_fact(fact_id: str, req: VerifyRequest, db: Session = Depends(get_db)):
    """Health worker confirms a fact as correct."""
    fact = db.query(ExtractedFact).filter(ExtractedFact.id == fact_id).first()
    if not fact:
        raise HTTPException(status_code=404, detail="Fact not found")

    # Update status
    fact.status = "verified"

    # Add to verified_by list
    verified_list = json.loads(fact.verified_by or "[]")
    if req.source_type not in verified_list:
        verified_list.append(req.source_type)
    fact.verified_by = json.dumps(verified_list)

    # Boost confidence
    fact.confidence = min(0.99, fact.confidence + 0.10)

    # Log the verification
    verification = Verification(
        fact_id=fact_id,
        source_type=req.source_type,
        value=req.value,
    )
    db.add(verification)
    db.commit()

    return {"status": "verified", "confidence": fact.confidence}


@router.post("/{fact_id}/reject")
def reject_fact(fact_id: str, db: Session = Depends(get_db)):
    """Health worker marks a fact as incorrect."""
    fact = db.query(ExtractedFact).filter(ExtractedFact.id == fact_id).first()
    if not fact:
        raise HTTPException(status_code=404, detail="Fact not found")

    fact.status = "unverified"
    fact.confidence = max(0.1, fact.confidence - 0.20)
    db.commit()

    return {"status": "unverified", "confidence": fact.confidence}


@router.post("/{fact_id}/resolve-conflict")
def resolve_conflict(fact_id: str, db: Session = Depends(get_db)):
    """Mark the chosen fact as correct, reject the conflicting one."""
    fact = db.query(ExtractedFact).filter(ExtractedFact.id == fact_id).first()
    if not fact:
        raise HTTPException(status_code=404, detail="Fact not found")

    # Mark this fact as verified
    fact.status = "verified"
    fact.conflict_details = None
    fact.confidence = 0.95

    # Mark conflicting facts as rejected
    if fact.conflict_details:
        conflicts = json.loads(fact.conflict_details)
        for c in conflicts:
            conflicting = db.query(ExtractedFact).filter(
                ExtractedFact.id == c.get("existing_fact_id")
            ).first()
            if conflicting:
                conflicting.status = "unverified"
                conflicting.confidence = 0.3

    db.commit()
    return {"status": "resolved"}

import React, { useRef, useState } from "react";
import { useParams, useNavigate } from "react-router-dom";

function CameraCapture() {
  const { id } = useParams();

  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);
  const [visibleFields, setVisibleFields] = useState([]);

  const startCamera = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: "environment" },  // back camera
      });
      videoRef.current.srcObject = stream;
      setStreaming(true);
    } catch (err) {
      alert("Camera access denied. Please allow camera permission.");
    }
  };

  const captureAndExtract = async () => {
    setLoading(true);
    setResult(null);
    setVisibleFields([]);

    // Capture frame from video
    const canvas = document.createElement("canvas");
    canvas.width = videoRef.current.videoWidth;
    canvas.height = videoRef.current.videoHeight;
    canvas.getContext("2d").drawImage(videoRef.current, 0, 0);

    canvas.toBlob(async (blob) => {
      const formData = new FormData();
      formData.append("file", blob, "capture.jpg");

      try {
        const res = await API.post(`/records/upload/${id}`, formData);
        setResult(res.data);

        const fields = Object.entries(res.data.extracted_data).filter(
          ([key]) => !key.startsWith("_")
        );
        for (let i = 0; i < fields.length; i++) {
          await new Promise((r) => setTimeout(r, 350));
          setVisibleFields((prev) => [...prev, fields[i]]);

        alert("Extraction failed: " + err.message);

MFRC522 mfrc522(SS_PIN, RST_PIN);
MFRC522::MIFARE_Key key;

// Change this to your patient's URL
String patientURL = "http://localhost:3000/patient/YOUR_PATIENT_ID_HERE";

void setup() {
  Serial.begin(115200);
  SPI.begin();
  mfrc522.PCD_Init();

  // Default key for NTAG
  for (byte i = 0; i < 6; i++) key.keyByte[i] = 0xFF;

  Serial.println("=================================");
  Serial.println("  CareLink NFC Wristband Writer");
  Serial.println("=================================");
  Serial.println("Hold an NFC wristband near the reader...");
}

void loop() {
  // Look for new cards
  if (!mfrc522.PICC_IsNewCardPresent()) return;
  if (!mfrc522.PICC_ReadCardSerial()) return;

  Serial.print("Tag detected! UID: ");
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    Serial.print(mfrc522.uid.uidByte[i] < 0x10 ? " 0" : " ");
    Serial.print(mfrc522.uid.uidByte[i], HEX);
  }
  Serial.println();

  // Write the URL to the tag (simplified — writes raw bytes)
  // For NTAG215, data starts at page 4
  byte pageData[4];
  byte urlBytes[48];
  int len = patientURL.length();

  // Copy URL to byte array
  for (int i = 0; i < len && i < 48; i++) {
    urlBytes[i] = patientURL[i];
  }

  // Write 4 bytes at a time to NTAG pages
  for (int page = 4; page < 4 + (len / 4) + 1 && page < 40; page++) {
    int offset = (page - 4) * 4;
    for (int i = 0; i < 4; i++) {
      pageData[i] = (offset + i < len) ? urlBytes[offset + i] : 0;
    }
    mfrc522.MIFARE_Ultralight_Write(page, pageData, 4);
  }

  Serial.println("✅ URL written to wristband!");
  Serial.println("URL: " + patientURL);
  Serial.println("Hold a different wristband or reset...");

  mfrc522.PICC_HaltA();
  delay(3000);
}
